\section{Ferramentas usadas}

\subsection*{Variáveis aleatórias mistas}

Uma VA é \emph{mista} quando possui parte discreta e parte contínua. Isso se
manifesta assim:
\begin{itemize}[nosep,leftmargin=*]
\item a \textbf{CDF} tem descontinuidades de salto e trechos contínuos crescentes;
\item a \textbf{PDF} tem impulsos e trechos de densidade finita.
\end{itemize}

As duas quantidades básicas, para qualquer ponto $a$:
\[
F_X(a^+) = \Pr[X \le a], \qquad F_X(a^-) = \Pr[X < a],
\qquad \Pr[X=a] = F_X(a^+) - F_X(a^-).
\]

Para intervalos, na notação de limites com $\pm$:
\[
\Pr[a < X \le b] = \int_{a^+}^{b^+} f_X \dd x, \qquad
\Pr[a \le X \le b] = \int_{a^-}^{b^+} f_X \dd x,
\]
\[
\Pr[a < X < b] = \int_{a^+}^{b^-} f_X \dd x, \qquad
\Pr[a \le X < b] = \int_{a^-}^{b^-} f_X \dd x .
\]

Em linguagem de CDF: \emph{no limite superior copie o sinal; no limite inferior
troque o sinal}. Isto é, $\Pr[a<X\le b]=F_X(b^+)-F_X(a^+)$ e
$\Pr[a\le X\le b]=F_X(b^+)-F_X(a^-)$.

\subsection*{Função impulso}

\[
\int_{-\infty}^{\infty}\!\!\delta(x)\dd x = \int_{0^-}^{0^+}\!\!\delta(x)\dd x = 1,
\qquad
\int_{-\infty}^{x}\!\!\delta(v)\dd v = u(x),
\qquad
u'(x) = \delta(x).
\]

\begin{resp}
\textbf{Propriedade central:} a derivada de uma descontinuidade de salto de
tamanho $c$ no ponto $x_0$ é um impulso de área $c$, isto é, $c\,\delta(x-x_0)$.
\end{resp}

Consequência prática, usada em todas as questões: \emph{salto na CDF $\Leftrightarrow$
átomo de probabilidade $\Leftrightarrow$ impulso na PDF}.

\subsection*{Momentos}

Teorema fundamental do valor esperado (permite trabalhar direto com $f_X$, sem
construir $f_Y$):
\[
\E[g(X)] = \int_{-\infty}^{\infty} g(x)\,f_X(x)\dd x .
\]

Fórmula alternativa da variância: $\var[X] = \E[X^2] - \E[X]^2$.

\subsection*{Distribuições conjuntas}

Marginais: $f_X(x)=\int f_{X,Y}(x,y)\dd y$ e $f_Y(y)=\int f_{X,Y}(x,y)\dd x$
(no caso discreto, somar linhas e colunas da tabela).
Condicional: $p_{X\mid Y}(x\mid y) = p_{X,Y}(x,y)/p_Y(y)$.
Independência: $f_{X,Y}(x,y)=f_X(x)f_Y(y)$ para \emph{todo} par $(x,y)$.
Covariância: $\cov[X,Y]=\E[XY]-\E[X]\E[Y]$.

\subsection*{(b) CDF}

Integrando caso a caso, no formato $F_X(x)=\int_{-\infty}^{x^+} f_X(u)\dd u$:

\medskip
\noindent\textbf{Caso $x<-1$:} região sem densidade nem impulso, $F_X(x)=0$.

\medskip
\noindent\textbf{Caso $-1\le x<1$:}
$\displaystyle F_X(x)=\int_{-1}^{x}\tfrac19\dd u=\tfrac{x+1}{9}$,
com $F_X(1^-)=\tfrac29$.

\medskip
\noindent\textbf{Caso $1\le x<2$:}
\[
F_X(x)=\underbrace{\int_{-1}^{1^-}\tfrac19\dd u}_{2/9}
+\underbrace{\int_{1^-}^{1^+}\tfrac1{18}\delta(u-1)\dd u}_{1/18}
+\underbrace{\int_{1}^{x}\tfrac29\dd u}_{\frac29(x-1)}
=\tfrac{5}{18}+\tfrac29(x-1),
\]
com $F_X(2^-)=\tfrac12$.

\medskip
\noindent\textbf{Caso $2\le x<3$:}
$F_X(x)=\tfrac12+\tfrac1{18}+\tfrac19(x-2)=\tfrac59+\tfrac{x-2}{9}$,
com $F_X(3^-)=\tfrac23$.

\medskip
\noindent\textbf{Caso $3\le x<4$:}
$F_X(x)=\tfrac23+\tfrac29+\tfrac19(x-3)=\tfrac89+\tfrac{x-3}{9}$,
com $F_X(4)=1$.

\medskip
\noindent\textbf{Caso $x\ge4$:} $F_X(x)=1$.

\begin{resp}
\textbf{Sumário (forma detalhada, com os pontos de salto isolados):}
\[
F_X(x)=\begin{cases}
0, & x<-1,\\[2pt]
\frac19(x+1), & -1\le x<1,\\[2pt]
\frac{5}{18}, & x=1,\\[2pt]
\frac{5}{18}+\frac29(x-1), & 1<x<2,\\[2pt]
\frac59, & x=2,\\[2pt]
\frac59+\frac19(x-2), & 2<x<3,\\[2pt]
\frac89, & x=3,\\[2pt]
\frac89+\frac19(x-3), & 3<x<4,\\[2pt]
1, & x\ge4.
\end{cases}
\qquad=\qquad
\begin{cases}
0, & x<-1,\\[2pt]
\frac19(x+1), & -1\le x<1,\\[2pt]
\frac{5}{18}+\frac29(x-1), & 1\le x<2,\\[2pt]
\frac59+\frac19(x-2), & 2\le x<3,\\[2pt]
\frac89+\frac19(x-3), & 3\le x<4,\\[2pt]
1, & 4\le x.
\end{cases}
\]
\end{resp}

A compactação à direita vale porque a CDF é \textbf{contínua à direita}: o valor no
ponto de salto coincide com o ramo da direita avaliado nesse ponto.

\subsection*{(c) Média}

Pelo teorema da probabilidade total aplicado à média:
\[
\E[X]=\underbrace{\tfrac52}_{A}\cdot\tfrac13+\underbrace{2}_{B}\cdot\tfrac16
+\underbrace{\tfrac12}_{C}\cdot\tfrac13+\underbrace{3}_{D}\cdot\tfrac16
=\tfrac56+\tfrac26+\tfrac16+\tfrac36 .
\]

\begin{resp}
$\displaystyle \E[X]=\frac{11}{6}\approx 1{,}833.$
\end{resp}

\subsection*{(d) $\Pr[X\ge2]$}

O sinal é $\ge$ e o ponto $2$ \emph{tem salto}, então é preciso usar $F_X(2^-)$:
\[
\Pr[X\ge2]=1-\Pr[X<2]=1-F_X(2^-)=1-\tfrac12 .
\]

\begin{resp}
$\displaystyle \Pr[X\ge2]=\frac12.$
\end{resp}

\emph{Atenção:} usar $F_X(2^+)=5/9$ por descuido daria $4/9$, que é $\Pr[X>2]$.
A diferença é exatamente o átomo $1/18$ em $x=2$.

\emph{Conferência independente} (teorema da probabilidade total):
$\tfrac23\cdot\tfrac13+\tfrac23\cdot\tfrac16+0\cdot\tfrac13+1\cdot\tfrac16
=\tfrac29+\tfrac19+\tfrac16=\tfrac12$. \checkmark

\section{Questão 2: tempo de espera no sinal de trânsito}

\textbf{Enunciado.} Sinal alterna 40\,s aberto e 20\,s fechado. $X$ é o tempo de
espera, em segundos, de um motorista que passa pelo sinal.

\subsection*{Modelagem}

O motorista chega num instante uniformemente distribuído no ciclo de $60$\,s.
Seja $U\sim\text{Uniform}([0,60])$ o instante de chegada medido a partir do início
da fase aberta, com densidade $1/60$. Então:
\begin{itemize}[nosep,leftmargin=*]
\item $U\in[0,40)$, com probabilidade $40/60=\tfrac23$: sinal aberto, $X=0$;
\item $U\in[40,60)$, com probabilidade $20/60=\tfrac13$: sinal fechado,
      $X=60-U$, uniforme entre $0$ e $20$.
\end{itemize}

$X$ é mista: átomo em zero com $2/3$ da probabilidade, mais parte contínua em
$(0,20]$ com o $1/3$ restante.

\subsection*{(a) PDF}

Parte discreta: $\Pr[X=0]=\tfrac23$, logo o termo $\tfrac23\delta(x)$.
Parte contínua: densidade $\tfrac13\cdot\tfrac1{20}=\tfrac1{60}$ em $[0,20]$
(equivalentemente, $X=60-U$ preserva a densidade $1/60$ de $U$).

\begin{resp}
\[
f_X(x)=\frac23\,\delta(x)+\frac1{60}\Ind{0\le x\le 20}.
\]
\end{resp}

\emph{Verificação:} $\tfrac23 + 20\cdot\tfrac1{60}=\tfrac23+\tfrac13=1$. \checkmark

\subsection*{(c) Tempo de espera médio}

\[
\E[X]=\int x f_X(x)\dd x
= \underbrace{0\cdot\tfrac23}_{\text{impulso}}
+ \int_0^{20}\frac{x}{60}\dd x
= \frac1{60}\left[\frac{x^2}{2}\right]_0^{20}
= \frac{200}{60}.
\]

\begin{resp}
$\displaystyle \E[X]=\frac{10}{3}\approx 3{,}33\ \text{segundos}.$
\end{resp}

\emph{Leitura:} quem pega o sinal fechado espera $10$\,s em média, mas isso ocorre
em apenas $1/3$ das passagens; daí $\tfrac13\cdot10=\tfrac{10}3$.
Se quiser, $\var[X]=\tfrac{400}{9}-\left(\tfrac{10}{3}\right)^2=\tfrac{100}{3}\approx33{,}3\ \mathrm{s^2}$.

\section{Questão 3: ceifador aplicado a uma Laplaciana}

\textbf{Enunciado.} $f_X(x)=\tfrac12 e^{-|x|}$ e $Y=g(X)$, com
$g(x)=-2$ se $x<-2$; $g(x)=x$ se $-2\le x\le2$; $g(x)=+2$ se $x>2$.

\subsection*{O que $g$ faz}

$g$ é um \emph{ceifador} (saturação / \emph{clipping}): deixa $X$ passar intacto
dentro de $[-2,2]$ e trava nos limites fora dela. Consequência: cada cauda de $X$,
que é um intervalo infinito, colapsa num \textbf{único ponto}, gerando um átomo.
$Y$ é, portanto, mista.

\end{tikzpicture}

\subsection*{Pesos dos átomos}

Para $x<0$ vale $-|x|=x$, logo $f_X(x)=\tfrac12 e^{x}$ nessa região (exponencial
crescente, cuja primitiva é $\tfrac12 e^{x}$, sem sinal negativo). Assim:
\[
\Pr[Y=-2]=\Pr[X<-2]=\int_{-\infty}^{-2}\tfrac12 e^{x}\dd x
=\tfrac12\Big[e^{x}\Big]_{-\infty}^{-2}=\tfrac12 e^{-2},
\]
\[
\Pr[Y=+2]=\Pr[X>2]=\int_{2}^{\infty}\tfrac12 e^{-x}\dd x
=\tfrac12\Big[-e^{-x}\Big]_{2}^{\infty}=\tfrac12 e^{-2}.
\]
Numericamente, $\tfrac12 e^{-2}\approx 0{,}0677$ em cada borda.

\subsection*{PDF de $Y$}

No miolo, $g$ é a identidade, então $\Pr[Y\in A]=\Pr[X\in A]$ para todo
$A\subset(-2,2)$ e a densidade é \emph{copiada sem alteração}. Formalmente, para
$-2<y<2$ tem-se $F_Y(y)=\Pr[X\le y]=F_X(y)$, e derivando, $f_Y(y)=f_X(y)$.

\begin{resp}
\[
f_Y(y)=\tfrac12 e^{-|y|}\Ind{-2<y<2}
+\tfrac12 e^{-2}\,\delta(y+2)
+\tfrac12 e^{-2}\,\delta(y-2).
\]
\end{resp}

\emph{Verificação de área:} $\int_{-2}^{2}\tfrac12 e^{-|y|}\dd y = 1-e^{-2}$,
mais $2\cdot\tfrac12 e^{-2}=e^{-2}$; total $=1$. \checkmark

\subsection*{Variância}

\textbf{Média.} $f_X$ é par e $g$ é ímpar, logo $Y$ e $-Y$ têm a mesma
distribuição e portanto $\E[Y]=0$. Explicitamente: o termo do miolo é a integral de
uma função ímpar em domínio simétrico, e os dois impulsos contribuem
$(-2)\tfrac12 e^{-2}+(+2)\tfrac12 e^{-2}=0$.

Com $\E[Y]=0$, tem-se $\var[Y]=\E[Y^2]$. Usando o teorema fundamental do valor
esperado direto em $f_X$ (sem precisar de $f_Y$):
\[
\E[Y^2]=\int_{-\infty}^{\infty} g(x)^2 f_X(x)\dd x
=\underbrace{\int_{-\infty}^{-2}\!\!4\cdot\tfrac12 e^{x}\dd x}_{2e^{-2}}
+\underbrace{\int_{-2}^{2}\!x^2\cdot\tfrac12 e^{-|x|}\dd x}_{2-10e^{-2}}
+\underbrace{\int_{2}^{\infty}\!\!4\cdot\tfrac12 e^{-x}\dd x}_{2e^{-2}} .
\]
A integral do meio, por paridade e integrando por partes com primitiva
$-(x^2+2x+2)e^{-x}$:
\[
\int_{-2}^{2}x^2\cdot\tfrac12 e^{-|x|}\dd x=\int_0^2 x^2 e^{-x}\dd x
=\Big[-(x^2+2x+2)e^{-x}\Big]_0^2=2-10e^{-2}.
\]

\begin{resp}
\[
\E[Y^2]=2e^{-2}+\left(2-10e^{-2}\right)+2e^{-2}=2-6e^{-2},
\qquad
\boxed{\var[Y]=2-6e^{-2}\approx 1{,}188 } .
\]
\end{resp}

\emph{Erro comum:} calcular apenas a integral do miolo ($2-10e^{-2}\approx0{,}647$)
e esquecer os dois impulsos, que contribuem $4e^{-2}\approx0{,}541$. Numa PDF mista
o momento tem que percorrer \emph{todas} as parcelas.

\emph{Leitura:} a Laplaciana original tem $\var[X]=2$; o ceifador derruba a
variância para $\approx1{,}19$ (redução de $41\%$). Apesar de as caudas terem
apenas $e^{-2}\approx13{,}5\%$ da probabilidade, elas respondiam por quase metade
da variância, porque os desvios entram ao quadrado.

\section{Questão 5: densidades conjuntas e independência}

\textbf{Método.} Marginal: integrar a conjunta na outra variável, com os limites
ditados pelo suporte. Independência: exigir $f_{X,Y}(x,y)=f_X(x)f_Y(y)$ para
\emph{todo} par $(x,y)$.

\subsection*{(a) $f_{X,Y}(x,y)=e^{-(x+y)}\Ind{x\ge0 \wedge y\ge0}$}

Suporte retangular (o quadrante). Fixado $x\ge0$, o $y$ varre $[0,\infty)$:
\[
f_X(x)=\int_0^{\infty} e^{-(x+y)}\dd y = e^{-x}\int_0^{\infty} e^{-y}\dd y = e^{-x},
\qquad x\ge0,
\]
e por simetria da expressão, $f_Y(y)=e^{-y}$ para $y\ge0$.

\begin{resp}
$f_X(x)=e^{-x}\Ind{x\ge0}$ e $f_Y(y)=e^{-y}\Ind{y\ge0}$, ambas
$\text{Exponencial}(1)$.\\[3pt]
Como $f_X(x)f_Y(y)=e^{-x}e^{-y}=e^{-(x+y)}=f_{X,Y}(x,y)$ em todo o plano,
\textbf{$X$ e $Y$ são independentes}.
\end{resp}

\subsection*{(b) $f_{X,Y}(x,y)=2e^{-(x+y)}\Ind{0\le x\le y}$}

Agora o suporte é o \textbf{triângulo} $0\le x\le y$, e os limites passam a depender
da outra variável.

\medskip
\noindent\textbf{Marginal de $X$.} Fixado $x\ge0$, a condição $x\le y$ obriga $y$ a
ir de $x$ até $\infty$:
\[
f_X(x)=\int_{x}^{\infty} 2e^{-(x+y)}\dd y
=2e^{-x}\Big[-e^{-y}\Big]_{x}^{\infty}
=2e^{-x}\cdot e^{-x}=2e^{-2x}, \qquad x\ge0 .
\]

\noindent\textbf{Marginal de $Y$.} Fixado $y\ge0$, a condição $0\le x\le y$ obriga
$x$ a ir de $0$ até $y$:
\[
f_Y(y)=\int_{0}^{y} 2e^{-(x+y)}\dd x
=2e^{-y}\Big[-e^{-x}\Big]_{0}^{y}
=2e^{-y}\left(1-e^{-y}\right), \qquad y\ge0 .
\]

\begin{resp}
$f_X(x)=2e^{-2x}\Ind{x\ge0}$ (ou seja, $X\sim\text{Exponencial}(2)$) e
$f_Y(y)=2e^{-y}(1-e^{-y})\Ind{y\ge0}$.\\[3pt]
\textbf{$X$ e $Y$ NÃO são independentes.}
\end{resp}

\noindent\textbf{Justificativa rápida (suporte).} Se o suporte não é um retângulo,
não pode haver independência. No ponto $(x,y)=(3,1)$:
$f_{X,Y}(3,1)=0$ (pois $3>1$), mas $f_X(3)f_Y(1)>0$. A igualdade falha, e nem foi
preciso calcular as marginais.

\noindent\textbf{Justificativa direta.}
$f_X(x)f_Y(y)=4e^{-2x-y}\left(1-e^{-y}\right)\ne 2e^{-(x+y)}$.
Por exemplo em $(0,1)$: $f_{X,Y}(0,1)=2e^{-1}\approx0{,}736$, enquanto
$f_X(0)f_Y(1)\approx0{,}930$.

\subsection*{Interpretação do item (b)}

Se $A,B\sim\text{Exponencial}(1)$ independentes e definirmos $X=\min(A,B)$ e
$Y=\max(A,B)$, a densidade conjunta é exatamente $2e^{-(x+y)}$ na região
$x\le y$ (o fator $2$ vem das duas ordens possíveis). Isso confirma as marginais:
o mínimo de duas $\text{Exp}(1)$ é $\text{Exp}(2)$, e o máximo tem CDF

evidentemente não são independentes.

\]
\end{resp}

\emph{Atalho geométrico} (vale por ser uniforme): $F_Y(y)$ é a área de $R$ abaixo da
altura $y$, dividida por $150$. Para $-10\le y\le0$ essa parte é um triângulo de
catetos $y+10$, área $(y+10)^2/2$; para $0\le y\le10$ é o triângulo inteiro ($50$)
mais um retângulo $10\times y$. Os mesmos resultados saem sem integrar.

\emph{Conferência por derivação:} $\frac{\dd}{\dd y}\frac{(y+10)^2}{300}=\frac{y+10}{150}$
\checkmark \; e $\frac{\dd}{\dd y}\left(\frac13+\frac{y}{15}\right)=\frac1{15}$ \checkmark.
Note que $F_Y$ é \textbf{contínua} (sem saltos), pois $Y$ é puramente contínua:
parábola no trecho negativo (densidade em rampa) e reta no positivo (densidade constante).

\subsection*{(e) Condicional $f_{Y\mid X}(y\mid X=5)$}

Condicionar em $X=5$ é olhar a fatia vertical naquele $x$: ali
$y$ vai de $5-10=-5$ até $10$, comprimento $15$. Como a conjunta é constante, a
condicional é \textbf{uniforme} nesse intervalo. Formalmente, com
$f_X(x)=\frac{20-x}{150}$ (fatia vertical de comprimento $20-x$), tem-se
$f_X(5)=\frac{15}{150}=\frac1{10}$ e
\[
f_{Y\mid X}(y\mid 5)=\frac{f_{X,Y}(5,y)}{f_X(5)}=\frac{1/150}{1/10}=\frac{1}{15}.
\]

\begin{resp}
$f_{Y\mid X}(y\mid 5)=\dfrac{1}{15}$ para $-5\le y\le 10$, e $0$ fora
(uniforme em $[-5,10]$).
\end{resp}

\subsection*{(f) Covariância}

\qquad
\E[Y]=\frac1{300}\int_0^{10}\!(20x-x^2)\dd x=\frac{20}{9},
\]

=\frac1{300}\cdot\frac{12500}{3}=\frac{125}{9}.

=\frac{1125}{81}-\frac{800}{81}=\frac{325}{81}\approx 4{,}012 .
\]
\end{resp}

\section{Questão 7: régua quebrada duas vezes}

\textbf{Enunciado.} Uma régua de comprimento unitário é quebrada em um ponto
aleatório; o pedaço da esquerda é novamente quebrado. $X$ é o primeiro ponto de
quebra e $Y$ o segundo, ambos medidos a partir da extremidade esquerda.

\subsection*{Modelagem: a chave da questão}

Diferente do usual, aqui não se recebe a conjunta: recebe-se um \textbf{experimento
em duas etapas}, e é dele que a conjunta é construída. Traduzindo:
\begin{resp}
\[
X\sim\text{Uniform}([0,1]),
\qquad
Y\mid X=x \;\sim\; \text{Uniform}([0,x]).
\]
\end{resp}
A segunda etapa só pode ser descrita \emph{condicionada} a $X$, porque o pedaço a
ser quebrado tem comprimento $X$. Por isso os itens vêm nessa ordem: marginal de
$X$, condicional de $Y$, e só então a conjunta.

\subsection*{(a) $f_X$}

Uniforme em $[0,1]$: \quad $f_X(x)=1$ para $0\le x\le1$ (retângulo de altura $1$).

\subsection*{(b) $f_{Y\mid X}(y\mid X=x)$}

Uniforme em $[0,x]$: \quad $f_{Y\mid X}(y\mid x)=\dfrac1x$ para $0\le y\le x$.

Note que \textbf{a altura muda com $x$}: para $x=1/4$ é um retângulo de altura $4$
sobre $[0,1/4]$; para $x=3/4$, altura $4/3$ sobre $[0,3/4]$. Em ambos a área é $1$.
Densidade maior para pedaço menor é esperado: a mesma probabilidade tem de caber
num intervalo mais curto.

\subsection*{(c) Conjunta $f_{X,Y}$}

Pela regra do produto, $f_{X,Y}(x,y)=f_X(x)\,f_{Y\mid X}(y\mid x)$:
\[
f_{X,Y}(x,y)=1\cdot\Ind{0\le x\le1}\cdot\frac1x\Ind{0\le y\le x}.
\]

\begin{resp}
\[
f_{X,Y}(x,y)=\frac1x\,\Ind{0\le y\le x\le 1}.
\]
Suporte: o triângulo de vértices $(0,0)$, $(1,0)$ e $(1,1)$, isto é, a metade do
quadrado unitário \emph{abaixo} da reta $y=x$ (pois a segunda quebra é sempre à
esquerda da primeira).
\end{resp}

\emph{Verificação:}
$\int_0^1\!\int_0^x \frac1x \dd y\dd x=\int_0^1 1 \dd x=1$. \checkmark

\subsection*{(d) Marginal $f_Y$}

\textbf{De onde saem os limites.} Os limites vêm do suporte, lidos com a outra
variável congelada. Congelando $y$ e quebrando a cadeia $0\le y\le x\le1$ nas
partes que envolvem $x$: de $y\le x$ vem $x\ge y$; de $x\le 1$ vem $x\le 1$. Logo
$x$ vai \textbf{de $y$ a $1$}. Geometricamente: a reta \emph{horizontal} na altura
$y$ entra no triângulo pela hipotenusa ($x=y$) e sai pela aresta direita ($x=1$).
\[
f_Y(y)=\int_{y}^{1}\frac1x\dd x=\Big[\ln x\Big]_{y}^{1}=-\ln y .
\]

\begin{resp}
$f_Y(y)=\ln(1/y)$ para $0<y\le1$.
\end{resp}

\emph{Cuidado com a direção da fatia:} para a marginal de $X$ congela-se $x$ e a
fatia é \emph{vertical}, com $y$ de $0$ a $x$; de fato
$f_X(x)=\int_0^x\frac1x\dd y=1$, recuperando o item (a). Teste rápido: os limites
nunca podem conter a variável que está sendo integrada.

\emph{Sobre o $1/x$:} ele é a \textbf{altura da superfície} $f_{X,Y}$ sobre o ponto
$(x,y)$. Ao longo da fatia horizontal o $x$ muda, logo a altura muda (vale $1/y$ na
entrada e $1$ na saída), e por isso a integral não é um produto simples: a primitiva
de $1/x$ é $\ln x$, daí o logaritmo. No item (b), por outro lado, $1/x$ era a altura
\emph{constante} de um retângulo, porque ali $x$ é parâmetro e a função é de $y$.

A densidade $\ln(1/y)$ explode quando $y\to0^+$ e vale $0$ em $y=1$, o que faz
sentido: a segunda quebra tende a ficar perto da extremidade esquerda. Densidade
infinita não é problema desde que a área seja finita: $\int_0^1\ln(1/y)\dd y=1$. \checkmark

\end{resp}

\subsection*{(f) Médias}

\qquad

\]
ou, por condicionamento (o mais rápido): $\E[Y\mid X]=X/2$, logo
$\E[Y]=\E[X/2]=\frac14$.

$\mu_X=\dfrac12$ \qquad e \qquad $\mu_Y=\dfrac14$.
\end{resp}

\subsection*{(g) Variâncias}

$\var[X]=\dfrac1{12}\approx0{,}0833$ \qquad e \qquad

=\int_0^1\!\!\int_0^x y \dd y\dd x=\int_0^1\frac{x^2}{2}\dd x=\frac16 .
\]
(O $x$ cancela com o $1/x$, o que torna a conta trivial.)

\begin{resp}
\[
\cov[X,Y]=\frac16-\frac12\cdot\frac14=\frac{1}{24},
\qquad

\end{resp}

Correlação positiva e forte: quebrar mais à direita na primeira vez libera espaço

\subsection*{(i) Probabilidade de os três pedaços formarem um triângulo}

\textbf{Passo 1, os pedaços.} Como $Y\le X$, os cortes em ordem são $0,Y,X,1$ e os
comprimentos são
\[
a=Y,\qquad b=X-Y,\qquad c=1-X .
\]

\textbf{Passo 2, a condição.} Como $a+b+c=1$, a desigualdade triangular $a<b+c$
equivale a $a<\frac12$; idem para os outros. Portanto:
\emph{formam triângulo se e somente se nenhum pedaço passa de $1/2$}, isto é,
\[
Y<\tfrac12,\qquad Y>X-\tfrac12,\qquad X>\tfrac12 .
\]

\textbf{Passo 3, a região.} Para $x\in(\frac12,1)$, o $y$ vai de $x-\frac12$ a
$\frac12$, comprimento $1-x$.

\textbf{Passo 4, integrar.}
\[
\Pr[\triangle]=\int_{1/2}^{1}\!\!\int_{x-1/2}^{1/2}\frac1x\dd y\dd x
=\int_{1/2}^{1}\frac{1-x}{x}\dd x
=\int_{1/2}^{1}\left(\frac1x-1\right)\dd x
=\Big[\ln x-x\Big]_{1/2}^{1}.
\]

\begin{resp}
\[
\Pr[\text{triângulo}]=(0-1)-\left(-\ln 2-\tfrac12\right)=\ln 2-\frac12\approx 19{,}31\% .
\]
\end{resp}

Menos de um em cinco. A intuição de que "quase sempre dá triângulo" falha por causa
do pedaço $c=1-X$: ele já passa de $1/2$ sempre que $X<\frac12$, ou seja, em metade
dos casos logo de saída.

\section{Questão 8: vetor aleatório de retransmissões}

\textbf{Enunciado.} Um terminal tem três mensagens a transmitir. Após enviar uma,
espera confirmação; se não vier, reenvia. Cada transmissão tem probabilidade $p$
de sucesso, independentemente das demais. $N_i$ é o número \emph{total} de
transmissões no instante em que a mensagem $i$ é recebida com sucesso.

\subsection*{Modelagem}

Seja $G_i$ o número de transmissões gastas \emph{apenas} com a mensagem $i$.
Como cada tentativa é um ensaio de Bernoulli$(p)$ independente e paramos no
primeiro sucesso, $G_i\sim\text{Geométrica}(p)$, com
$\Pr[G_i=g]=p(1-p)^{g-1}$ para $g\ge1$, e $G_1,G_2,G_3$ independentes.

Como $N_i$ conta o total \emph{acumulado}:
\begin{resp}
\[
N_1=G_1,\qquad N_2=G_1+G_2,\qquad N_3=G_1+G_2+G_3 .
\]
Equivalentemente, $G_1=N_1$, $G_2=N_2-N_1$ e $G_3=N_3-N_2$: o vetor $\vec N$ é a
\textbf{soma acumulada} de três geométricas independentes.
\end{resp}

Daí já sai a restrição de suporte: os $G_i$ valem pelo menos $1$, logo
\[
1\le N_1<N_2<N_3 ,
\]
com desigualdades \emph{estritas}, pois cada mensagem consome ao menos uma
transmissão.

\subsection*{(a) PMF conjunta de $\vec N$}

O evento $\{N_1=n_1, N_2=n_2, N_3=n_3\}$ é exatamente o evento
$\{G_1=n_1,\; G_2=n_2-n_1,\; G_3=n_3-n_2\}$. Por independência, multiplicam-se as
três geométricas:
\[
p_{\vec N}(n_1,n_2,n_3)
= \underbrace{p(1-p)^{n_1-1}}_{G_1}
\cdot \underbrace{p(1-p)^{n_2-n_1-1}}_{G_2}
\cdot \underbrace{p(1-p)^{n_3-n_2-1}}_{G_3}.
\]
Os três fatores $p$ viram $p^3$ e os expoentes de $(1-p)$ se telescopam:
\[
(n_1-1)+(n_2-n_1-1)+(n_3-n_2-1) = n_3-3 .
\]

\begin{resp}
\[
p_{\vec N}(n_1,n_2,n_3)=p^3\,(1-p)^{\,n_3-3}\;\Ind{1\le n_1<n_2<n_3}.
\]
\end{resp}

\emph{Leitura:} a conjunta \textbf{não depende} de $n_1$ e $n_2$ isoladamente, só
de $n_3$. Faz sentido: dado que o total foi $n_3$, houve exatamente $3$ sucessos e
$n_3-3$ falhas, e a ordem em que as falhas se distribuíram entre as três mensagens
não altera a probabilidade. Todos os arranjos compatíveis são equiprováveis.

\subsection*{(b) Marginais}

\noindent\textbf{$N_1$.} É a própria primeira geométrica:
\[
p_{N_1}(n)=p(1-p)^{n-1}\Ind{n\ge1}.
\]

\noindent\textbf{$N_2$.} Soma de duas geométricas independentes, ou seja, uma
\emph{Pascal de ordem 2} (binomial negativa): o evento $\{N_2=n\}$ significa
``o segundo sucesso ocorreu na $n$-ésima transmissão''. Isso exige $1$ sucesso nas
$n-1$ primeiras (de $\binom{n-1}{1}$ maneiras) e sucesso na $n$-ésima:
\[
p_{N_2}(n)=\binom{n-1}{1}p^2(1-p)^{n-2}=(n-1)\,p^2(1-p)^{n-2}\Ind{n\ge2}.
\]

\noindent\textbf{$N_3$.} Mesmo raciocínio com o terceiro sucesso na $n$-ésima
transmissão, escolhendo as posições dos $2$ sucessos anteriores entre as $n-1$
primeiras:
\[
p_{N_3}(n)=\binom{n-1}{2}p^3(1-p)^{n-3}=\frac{(n-1)(n-2)}{2}\,p^3(1-p)^{n-3}\Ind{n\ge3}.
\]

\begin{resp}
\[
p_{N_1}(n)=p(1-p)^{n-1}\Ind{n\ge1},\quad
p_{N_2}(n)=(n-1)p^2(1-p)^{n-2}\Ind{n\ge2},
\]
\[
p_{N_3}(n)=\tfrac{(n-1)(n-2)}{2}\,p^3(1-p)^{n-3}\Ind{n\ge3}.
\]
\end{resp}

\emph{Pelo somatório da conjunta}, o mesmo resultado aparece como contagem. Por
exemplo, para $N_3$:
\[
p_{N_3}(n_3)=\sum_{n_1<n_2<n_3} p^3(1-p)^{n_3-3}
= p^3(1-p)^{n_3-3}\cdot\#\{(n_1,n_2): 1\le n_1<n_2<n_3\},
\]
e o número de pares é $\binom{n_3-1}{2}$. Como o somando é constante, a marginal é
literalmente ``constante vezes contagem'', o que reforça a leitura do item (a).

\emph{Médias}, se pedirem: $\E[N_i]=i/p$, pois cada geométrica tem média $1/p$.

\section{Questão 9: XOR e independência par-a-par}

\textbf{Enunciado.} $A,B\sim\text{Bernoulli}(1/2)$ independentes e
$C=A\oplus B$ (soma módulo 2). Seja $\vec X=[A\;B\;C]^{\mathsf T}$.

\subsection*{(a) PMF conjunta}

Os pares $(A,B)$ são equiprováveis, cada um com probabilidade $1/4$, e $C$ fica
\emph{determinado} por eles. Enumerando:

\begin{center}
\begin{tabular}{cccc}
\toprule
$a$ & $b$ & $c=a\oplus b$ & $\Pr$ \\
\midrule
0 & 0 & 0 & $1/4$ \\
0 & 1 & 1 & $1/4$ \\
1 & 0 & 1 & $1/4$ \\
1 & 1 & 0 & $1/4$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{resp}
\[
p_{\vec X}(a,b,c)=\tfrac14\,\Ind{c=a\oplus b},\qquad a,b,c\in\{0,1\}.
\]
Dos $8$ pontos do cubo $\{0,1\}^3$, apenas $4$ têm probabilidade não nula (os de
\emph{paridade par}); os outros $4$ têm probabilidade zero.
\end{resp}

\subsection*{(b) Marginais}

Somando a conjunta sobre as demais variáveis, cada valor aparece em duas das
quatro linhas, logo com probabilidade $2\times\frac14=\frac12$:

\begin{resp}
$A\sim\text{Bernoulli}(1/2)$, \quad $B\sim\text{Bernoulli}(1/2)$, \quad
$C\sim\text{Bernoulli}(1/2)$.
\end{resp}

Para $C$ vale também o argumento direto:
$\Pr[C=1]=\Pr[A\ne B]=\Pr[(0,1)]+\Pr[(1,0)]=\tfrac14+\tfrac14=\tfrac12$.

\subsection*{(c) Independência par-a-par}

Basta verificar $p_{U,V}(u,v)=p_U(u)\,p_V(v)$ para os três pares. Como todas as
marginais valem $1/2$, o produto é sempre $1/4$; falta ver que cada conjunta de
duas variáveis também vale $1/4$ nos quatro pontos.

\noindent\textbf{$A$ e $B$:} independentes por hipótese do enunciado, e de fato
$p_{A,B}(a,b)=1/4$ para os quatro pares.

\noindent\textbf{$A$ e $C$:} fixando $a$, das duas linhas com aquele $a$ uma tem
$c=0$ e a outra $c=1$. Logo $p_{A,C}(a,c)=1/4$ para todo par, que é
$\tfrac12\cdot\tfrac12$. \checkmark

\noindent\textbf{$B$ e $C$:} idem, por simetria entre $A$ e $B$. \checkmark

\begin{center}
\begin{tabular}{c|cc|c}
\toprule
$p_{A,C}$ & $c=0$ & $c=1$ & $p_A$ \\
\midrule
$a=0$ & $1/4$ & $1/4$ & $1/2$ \\
$a=1$ & $1/4$ & $1/4$ & $1/2$ \\
\midrule
$p_C$ & $1/2$ & $1/2$ & $1$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{resp}
As três conjuntas duas a duas fatoram, portanto $A,B,C$ são
\textbf{independentes par-a-par}.
\end{resp}

\emph{Intuição:} saber apenas $A$ não diz nada sobre $C$, pois $B$ ainda é um
``segredo'' uniforme que embaralha o resultado. É o mesmo princípio da cifra de
Vernam (one-time pad): o texto cifrado, sozinho, não revela nada sobre a mensagem.

\subsection*{(d) Não há independência mútua}

Independência mútua exigiria
$p_{\vec X}(a,b,c)=p_A(a)\,p_B(b)\,p_C(c)=\tfrac18$ para \emph{todos} os oito
pontos. Basta um contraexemplo:
\[
p_{\vec X}(0,0,1)=0
\qquad\text{mas}\qquad
p_A(0)\,p_B(0)\,p_C(1)=\tfrac12\cdot\tfrac12\cdot\tfrac12=\tfrac18 .
\]

\begin{resp}
$0\ne\tfrac18$, logo $A,B,C$ \textbf{não são mutuamente independentes}.
\end{resp}

O motivo estrutural: conhecidas \emph{duas} das variáveis, a terceira fica
completamente determinada ($C=A\oplus B$, mas também $A=B\oplus C$ e
$B=A\oplus C$). A dependência é de ordem três: ela só se manifesta quando se
olham as três juntas.

\subsection*{(e) Vetor média e matriz covariância de $\vec X$}

Cada componente é Bernoulli$(1/2)$, logo $\E[\cdot]=\tfrac12$ e
$\var[\cdot]=\tfrac12\cdot\tfrac12=\tfrac14$. As covariâncias cruzadas são nulas,
pois independência par-a-par implica descorrelação par-a-par. Conferindo uma delas:
\[
\E[AC]=\Pr[A=1,C=1]=\tfrac14,
\qquad
\cov[A,C]=\tfrac14-\tfrac12\cdot\tfrac12=0 .
\]

\begin{resp}
\[
\vec\mu_{\vec X}=\begin{bmatrix}1/2\\1/2\\1/2\end{bmatrix},
\qquad
C_{\vec X}=\frac14\begin{bmatrix}1&0&0\\0&1&0\\0&0&1\end{bmatrix}.
\]
\end{resp}

\subsection*{(f) Comparação com três Bernoulli independentes}

Para $\vec X'=[A'\;B'\;C']^{\mathsf T}$ com as três independentes, cada uma
Bernoulli$(1/2)$: as médias são $1/2$, as variâncias $1/4$ e as covariâncias são
nulas por independência.

\begin{resp}
\[
\vec\mu_{\vec X'}=\begin{bmatrix}1/2\\1/2\\1/2\end{bmatrix},
\qquad
C_{\vec X'}=\frac14 I_3 ,
\]
\emph{exatamente iguais} aos do item (e).
\end{resp}

\textbf{Conclusão.} Os dois vetores têm o mesmo vetor média e a mesma matriz
covariância, mas distribuições conjuntas completamente diferentes: $\vec X$ vive
em apenas $4$ dos $8$ vértices do cubo, enquanto $\vec X'$ ocupa os $8$.

Isso mostra que \textbf{média e covariância não caracterizam a distribuição
conjunta}. Elas capturam apenas estatísticas de \emph{segunda ordem}, isto é,
relações lineares duas a duas. A dependência do XOR é de terceira ordem e passa
despercebida por completo.

É o mesmo alerta da seção ``Descorrelação'' do slide \texttt{pre-2-2}:
\emph{independência implica descorrelação, mas a recíproca é falsa}. Aqui, as
variáveis são descorrelacionadas e até independentes par-a-par, e mesmo assim
conhecer duas delas determina a terceira sem nenhuma incerteza.

\section{Questão 10: vetores de produtos e de somas acumuladas}

\textbf{Enunciado.} $X_1,X_2,X_3 \overset{\text{i.i.d.}}{\sim}\text{Uniform}(-1,2)$.
Em (a), $Y_1=X_1$, $Y_2=X_1X_2$, $Y_3=X_1X_2X_3$; em (b),
$Z_1=Y_1$, $Z_2=Y_1+Y_2$, $Z_3=Y_1+Y_2+Y_3$, pela formulação matricial.

\subsection*{Os dois momentos que resolvem tudo}

Com $a=-1$ e $b=2$, a densidade é $f_X(x)=\frac13$ em $[-1,2]$ e, pela definição,
\[
\E[X]=\int_{-1}^{2}\frac{x}{3}\dd x=\frac13\left[\frac{x^2}{2}\right]_{-1}^{2}
=\frac13\left(2-\tfrac12\right)=\frac12,
\]
\[
\E[X^2]=\int_{-1}^{2}\frac{x^2}{3}\dd x=\frac13\left[\frac{x^3}{3}\right]_{-1}^{2}
=\frac13\left(\tfrac83+\tfrac13\right)=1 .
\]
(Equivalentemente pelas fórmulas prontas: $\frac{a+b}{2}=\frac12$ e
$\var=\frac{(b-a)^2}{12}=\frac34$, donde $\E[X^2]=\frac34+\frac14=1$.)

\begin{resp}
$\E[X]=\tfrac12$ \quad e \quad $\E[X^2]=1$.
\end{resp}

O valor $\E[X^2]=1$ é o que torna a questão viável à mão: todo fator ao quadrado
simplesmente some das contas.

\subsection*{(a) Vetor média de $\vec Y$}

Por independência, a esperança do produto é o produto das esperanças:
\[
\E[Y_1]=\tfrac12,\qquad \E[Y_2]=\left(\tfrac12\right)^2=\tfrac14,
\qquad \E[Y_3]=\left(\tfrac12\right)^3=\tfrac18 .
\]

\begin{resp}
$\vec\mu_{\vec Y}=\left[\tfrac12\;\;\tfrac14\;\;\tfrac18\right]^{\mathsf T}$.
\end{resp}

\subsection*{(a) Matriz covariância de $\vec Y$}

Escreva $Y_iY_j$ em termos dos $X$ e conte os expoentes: cada $X$ ao quadrado
contribui $\E[X^2]=1$; cada $X$ sozinho contribui $\E[X]=\tfrac12$.

\begin{center}
\begin{tabular}{lccl}
\toprule
produto & em termos dos $X$ & expoentes & valor \\
\midrule
$Y_1^2$ & $X_1^2$ & $(2,0,0)$ & $1$ \\
$Y_2^2$ & $X_1^2X_2^2$ & $(2,2,0)$ & $1$ \\
$Y_3^2$ & $X_1^2X_2^2X_3^2$ & $(2,2,2)$ & $1$ \\
$Y_1Y_2$ & $X_1^2X_2$ & $(2,1,0)$ & $1\cdot\tfrac12=\tfrac12$ \\
$Y_1Y_3$ & $X_1^2X_2X_3$ & $(2,1,1)$ & $\tfrac12\cdot\tfrac12=\tfrac14$ \\
$Y_2Y_3$ & $X_1^2X_2^2X_3$ & $(2,2,1)$ & $\tfrac12$ \\
\bottomrule
\end{tabular}
\end{center}

Subtraindo o produto das médias em cada caso:

\begin{resp}
\[
C_{\vec Y}=
\begin{bmatrix}
3/4 & 3/8 & 3/16\\[2pt]
3/8 & 15/16 & 15/32\\[2pt]
3/16 & 15/32 & 63/64
\end{bmatrix}
\]
\end{resp}

\emph{Fórmula fechada}, útil para conferir: para $i\le j$,
$\cov[Y_i,Y_j]=2^{\,i-j}-2^{-(i+j)}$. Exemplo: $i=2$, $j=3$ dá
$\tfrac12-\tfrac1{32}=\tfrac{15}{32}$. \checkmark

Todas as covariâncias são positivas porque os $Y_i$ \textbf{compartilham fatores}:
todos contêm $X_1$, e $Y_2,Y_3$ compartilham $X_1X_2$.

\subsection*{(b) Formulação matricial}

Escreva $\vec Z=A\,\vec Y$ com $A$ triangular inferior de uns:
\[
A=\begin{bmatrix}1&0&0\\1&1&0\\1&1&1\end{bmatrix},
\qquad
\vec\mu_{\vec Z}=A\,\vec\mu_{\vec Y},
\qquad
C_{\vec Z}=A\,C_{\vec Y}\,A^{\mathsf T}.
\]

\emph{Por que essa fórmula:} de $\vec Z-\vec\mu_{\vec Z}=A(\vec Y-\vec\mu_{\vec Y})$
segue
$\E\!\left[A(\vec Y-\vec\mu)(\vec Y-\vec\mu)^{\mathsf T}A^{\mathsf T}\right]
= A\,C_{\vec Y}\,A^{\mathsf T}$, pois $A$ é constante e sai da esperança. É o
análogo matricial de $\var[aX]=a^2\var[X]$.

\[
\vec\mu_{\vec Z}=\left[\tfrac12,\;\tfrac12+\tfrac14,\;\tfrac12+\tfrac14+\tfrac18\right]^{\mathsf T}
=\left[\tfrac12\;\;\tfrac34\;\;\tfrac78\right]^{\mathsf T}.
\]

\paragraph{Trabalhando em $1/64$.} Escrevendo $C_{\vec Y}$ com denominador comum
$64$, ela vira a matriz de inteiros $\begin{bmatrix}48&24&12\\24&60&30\\12&30&63\end{bmatrix}$.

\paragraph{Primeira multiplicação, $M=A\,C_{\vec Y}$.} A matriz da \emph{esquerda}
entra com suas \textbf{linhas}, que percorrem as \textbf{colunas} de $C_{\vec Y}$
(soma vertical). Linha $i$ de $A$ tem $1$ nas $i$ primeiras posições:

\begin{center}
\begin{tabular}{llll}
\toprule
linha de $A$ & entrada $(i,1)$ & $(i,2)$ & $(i,3)$ \\
\midrule
$[1,0,0]$ & $48$ & $24$ & $12$ \\
$[1,1,0]$ & $48{+}24=72$ & $24{+}60=84$ & $12{+}30=42$ \\
$[1,1,1]$ & $48{+}24{+}12=84$ & $24{+}60{+}30=114$ & $12{+}30{+}63=105$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Segunda multiplicação, $C_{\vec Z}=M\,A^{\mathsf T}$.} Agora $M$ está à
esquerda, então a \textbf{linha fica congelada} e percorrem-se as \textbf{colunas}
de $A^{\mathsf T}$, que são $[1,0,0]$, $[1,1,0]$ e $[1,1,1]$ (as colunas de
$A^{\mathsf T}$ são as linhas de $A$). O efeito é a soma acumulada \emph{de cada
linha}, da esquerda para a direita:

\begin{center}
\begin{tabular}{lll}
\toprule
linha de $M$ & soma acumulada & \\
\midrule
$48,\;24,\;12$ & $48 \to 72 \to 84$ & \\
$72,\;84,\;42$ & $72 \to 156 \to 198$ & \\
$84,\;114,\;105$ & $84 \to 198 \to 303$ & \\
\bottomrule
\end{tabular}
\end{center}

\begin{resp}
\[
\vec\mu_{\vec Z}=\begin{bmatrix}1/2\\3/4\\7/8\end{bmatrix},
\qquad
C_{\vec Z}=\frac1{64}\begin{bmatrix}48&72&84\\72&156&198\\84&198&303\end{bmatrix}
=\begin{bmatrix}
3/4 & 9/8 & 21/16\\[2pt]
9/8 & 39/16 & 99/32\\[2pt]
21/16 & 99/32 & 303/64
\end{bmatrix}
\]
\end{resp}

\subsection*{Leitura e conferências}

\textbf{Atalho.} Juntando as duas etapas (acumular por colunas, depois por linhas),
cada entrada de $C_{\vec Z}$ é a \textbf{soma do bloco superior esquerdo}
$i\times j$ de $C_{\vec Y}$:
\[
\cov[Z_i,Z_j]=\sum_{k\le i}\sum_{l\le j}\cov[Y_k,Y_l].
\]
Por exemplo $\cov[Z_2,Z_2]=48+24+24+60=156$ (em $1/64$), ou seja $39/16$; e
$\var[Z_3]$ é a soma das nove entradas, $303/64$.

\textbf{Conferência escalar.} A mesma entrada pela fórmula usual:
\[
\var[Z_2]=\var[Y_1+Y_2]=\var[Y_1]+\var[Y_2]+2\cov[Y_1,Y_2]
=\tfrac34+\tfrac{15}{16}+2\cdot\tfrac38=\tfrac{39}{16}. \checkmark
\]
As quatro parcelas são exatamente as quatro entradas do bloco $2\times2$: as
fórmulas escalares são a versão desenrolada da multiplicação matricial.

\textbf{Erro comum.} Trocar linha por coluna em $A$. Dois testes pegam isso na
hora: (i) $C_{\vec Z}$ tem de ser \emph{simétrica}; (ii) a entrada $(1,1)$ tem de
ser $\var[Z_1]=\var[Y_1]=3/4$. Além disso, as variâncias na diagonal têm de
\emph{crescer}, já que $Z_1$ soma um termo, $Z_2$ soma dois e $Z_3$ soma três
variáveis positivamente correlacionadas: $\tfrac34<\tfrac{39}{16}<\tfrac{303}{64}$.
\checkmark

\subsection*{4.7.6}
