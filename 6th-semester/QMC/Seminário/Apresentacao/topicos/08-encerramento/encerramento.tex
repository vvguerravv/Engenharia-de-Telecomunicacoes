\section*{Conclusão}

\begin{frame}{Conclusões}
  \begin{itemize}
    \item Semicondutor é um material cuja condução nós conseguimos
          \termo{controlar}.
    \item Esse controle vem da \termo{dopagem}, um átomo de boro ou de
          fósforo em cada milhão de átomos de silício.
    \item Por causa dessa dose, o silício precisa ser quase perfeitamente
          puro.
    \item No fim, o preço acompanha a \termo{pureza}, e não a quantidade de
          material.
  \end{itemize}
\end{frame}
