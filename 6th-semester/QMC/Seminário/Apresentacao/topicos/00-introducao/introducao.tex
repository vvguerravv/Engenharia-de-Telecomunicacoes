\section{Introdução}

\begin{frame}{Semicondutores nas telecomunicações}
  \begin{itemize}
    \item Celular, antena, satélite e roteador só funcionam por causa de
          alguns \termo{cristais semicondutores}.
    \item A matéria-prima deles é areia. O que transforma areia em chip é
          \alert{química}.
  \end{itemize}
  \vspace{0.4cm}
  \begin{block}{Ordem dos tópicos}
    Material \seta{} estrutura \seta{} pureza \seta{} ligação \seta{} reação
    \seta{} dispositivo \seta{} preço.
  \end{block}
\end{frame}
